cmake_minimum_required(VERSION 3.28)
project(wtv-poc CXX)
set(CMAKE_CXX_STANDARD 20)
# OBS_BUILD_DIR: directório de build do obs-studio (ver docs/INSTALACAO.md)
set(OBS_BUILD_DIR "$ENV{HOME}/.cache/wtv-obs/build" CACHE PATH "build do obs-studio")
set(OBS_SRC_DIR "$ENV{HOME}/.cache/wtv-obs/obs-studio" CACHE PATH "fontes do obs-studio")
set(OBS_FW "${OBS_BUILD_DIR}/libobs/RelWithDebInfo/libobs.framework")
add_executable(wtv-poc main.cpp)
target_include_directories(wtv-poc PRIVATE "${OBS_FW}/Headers" "${OBS_SRC_DIR}/libobs" "${OBS_BUILD_DIR}/config" "${OBS_SRC_DIR}/.deps/obs-deps-2025-08-23-universal/include")
target_link_libraries(wtv-poc PRIVATE "${OBS_FW}/libobs")
set_target_properties(wtv-poc PROPERTIES BUILD_RPATH "${OBS_BUILD_DIR}/libobs/RelWithDebInfo")
